static int FindNextMilestone(
  int[] years, int target)
{
  int low = 0, high = years.Length;
  while (low < high)
  {
    var mid = low + (high - low) / 2;
    if (years[mid] < target) low = mid + 1;
    else high = mid;
  }
  return low < years.Length ? years[low] : -1;
}
